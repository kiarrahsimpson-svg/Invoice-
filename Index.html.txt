// ── LOCAL STORAGE WRAPPERS ──
const LS = {
  get: k => { try { const v = localStorage.getItem(k); return v ? JSON.parse(v) : null; } catch { return null; } },
  set: (k,v) => { try { localStorage.setItem(k, JSON.stringify(v)); } catch {} },
  del: k => { try { localStorage.removeItem(k); } catch {} }
};

// ── AUTH ──
let currentUser = null;

function getUsers() { return LS.get('if_users') || {}; }
function saveUsers(u) { LS.set('if_users', u); }

function simpleHash(s) {
  let h = 5381;
  for (let i = 0; i < s.length; i++) h = ((h << 5) + h) ^ s.charCodeAt(i);
  return (h >>> 0).toString(36);
}

function doSignup() {
  const name = document.getElementById('signupName').value.trim();
  const email = document.getElementById('signupEmail').value.trim().toLowerCase();
  const pass = document.getElementById('signupPass').value;
  const errEl = document.getElementById('signupError');
  errEl.innerHTML = '';
  if (!name) return errEl.innerHTML = '<div class="m-err">Please enter your name.</div>';
  if (!email.includes('@')) return errEl.innerHTML = '<div class="m-err">Enter a valid email address.</div>';
  if (pass.length < 6) return errEl.innerHTML = '<div class="m-err">Password must be at least 6 characters.</div>';
  const users = getUsers();
  if (users[email]) return errEl.innerHTML = '<div class="m-err">An account with that email already exists.</div>';
  users[email] = { name, email, ph: simpleHash(pass), createdAt: Date.now() };
  saveUsers(users);
  loginUser({ name, email });
  closeModal();
  showToast(`Welcome, ${name}! 🎉`);
}

function doLogin() {
  const email = document.getElementById('loginEmail').value.trim().toLowerCase();
  const pass = document.getElementById('loginPass').value;
  const errEl = document.getElementById('loginError');
  errEl.innerHTML = '';
  const users = getUsers();
  const user = users[email];
  if (!user || user.ph !== simpleHash(pass)) return errEl.innerHTML = '<div class="m-err">Incorrect email or password.</div>';
  loginUser(user);
  closeModal();
  showToast(`Welcome back, ${user.name}!`);
}

function loginUser(user) {
  currentUser = user;
  LS.set('if_session', user.email);
  document.getElementById('authGate').classList.remove('active');
  document.getElementById('app').style.display = 'block';
  document.getElementById('userEmail').textContent = user.email;
  document.getElementById('userAvatar').textContent = (user.name || user.email)[0].toUpperCase();
  refreshHistory();
}

function doLogout() {
  currentUser = null;
  LS.del('if_session');
  closeModal();
  document.getElementById('app').style.display = 'none';
  document.getElementById('authGate').classList.add('active');
  switchTab('editor');
  newInvoice();
}

let editingId = null;
function uKey(s) { return `if_${currentUser.email}_${s}`; }
function getInvoices() { return LS.get(uKey('invoices')) || []; }
function setInvoices(list) { LS.set(uKey('invoices'), list); }

function collectData() {
  return {
    fromName: V('fromName'), fromTagline: V('fromTagline'),
    fromEmail: V('fromEmail'), fromAddress: V('fromAddress'),
    toName: V('toName'), toEmail: V('toEmail'), toAddress: V('toAddress'),
    invNumber: V('invNumber'), status: V('invStatus'),
    invDate: V('invDate'), dueDate: V('dueDate'),
    currency: V('currency'), taxRate: V('taxRate'),
    notes: V('notes'), payment: V('payment'),
    items: JSON.parse(JSON.stringify(items)),
    total: calcTotals().total
  };
}

function saveInvoice() {
  if (!currentUser) return showToast('Please sign in to save invoices.');
  const data = collectData();
  const list = getInvoices();
  const idx = list.findIndex(i => i.id === editingId);
  if (idx >= 0) {
    list[idx] = { ...data, id: editingId, savedAt: Date.now() };
    showToast('Invoice updated! ✓');
  } else {
    editingId = 'inv_' + Date.now();
    list.unshift({ ...data, id: editingId, savedAt: Date.now() });
    showToast('Invoice saved! ✓');
  }
  setInvoices(list);
  refreshHistory();
}

function loadInvoice(inv) {
  editingId = inv.id;
  items = (inv.items && inv.items.length) ? JSON.parse(JSON.stringify(inv.items)) : [{ desc:'', qty:1, rate:'' }];
  const fields = ['fromName','fromTagline','fromEmail','fromAddress','toName','toEmail','toAddress','invNumber','invDate','dueDate','taxRate','notes','payment'];
  const map = { invNumber:'invNumber', status:'invStatus', currency:'currency' };
  fields.forEach(f => { const el = document.getElementById(f); if (el && inv[f] !== undefined) el.value = inv[f]; });
  if (document.getElementById('invStatus')) document.getElementById('invStatus').value = inv.status || 'draft';
  if (document.getElementById('currency')) document.getElementById('currency').value = inv.currency || 'USD';
  renderItems(); render();
  switchTab('editor');
  showToast('Invoice loaded for editing.');
}

function deleteInvoice(id) {
  document.getElementById('confirmDeleteBtn').onclick = () => {
    setInvoices(getInvoices().filter(i => i.id !== id));

    closeModal();

  };
  showModal('delete');
}

function initEditor() {
  const today = new Date(), due = new Date(today);
  due.setDate(due.getDate() + 30);
  document.getElementById('invDate').value = toISO(today);
  document.getElementById('dueDate').value = toISO(due);
  document.getElementById('invNumber').value = 'INV-' + (Math.floor(Math.random()*900)+100);
  renderItems(); render();
}

function newInvoice() {
  editingId = null;
  items = [{ desc:'', qty:1, rate:'' }];
  ['fromName','fromTagline','fromEmail','fromAddress','toName','toEmail','toAddress','notes','payment'].forEach(id => {
    const el = document.getElementById(id); if (el) el.value = '';
  });
  document.getElementById('taxRate').value = '0';
  document.getElementById('currency').value = 'USD';
  document.getElementById('invStatus').value = 'draft';

}

function addItem() { items.push({ desc:'', qty:1, rate:'' }); renderItems(); render(); }
function removeItem(i) { if (items.length===1) return; items.splice(i,1); renderItems(); render(); }
function updateItem(i, f, val) { items[i][f] = val; render(); }

function renderItems() {
  document.getElementById('lineItems').innerHTML = items.map((item,i) => `
    <div class="line-item">

      <input type="number" placeholder="1" value="${item.qty}" min="0" step="any" oninput="updateItem(${i},'qty',this.value)" style="text-align:right"/>
      <input type="number" placeholder="0.00" value="${item.rate}" min="0" step="any" oninput="updateItem(${i},'rate',this.value)" style="text-align:right"/>

    </div>`).join('');

function V(id) { const el = document.getElementById(id); return el ? el.value : ''; }
function ea(s) { return (s||'').replace(/"/g,'&quot;'); }
function esc(s) { return (s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/\n/g,'<br>'); }
function toISO(d) { return d.toISOString().slice(0,10); }
function fmtD(str) {
  if (!str) return '—';
  const [y,m,d] = str.split('-');
  return ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'][parseInt(m)-1] + ' ' + parseInt(d) + ', ' + y;
}

function calcTotals() {
  const sym = currencySymbols[V('currency')] || '$';
  const tax = parseFloat(V('taxRate')) || 0;
  let sub = 0;
  items.forEach(it => { sub += (parseFloat(it.qty)||0) * (parseFloat(it.rate)||0); });
  const taxAmt = sub * tax / 100;
  const total = sub + taxAmt;
  const fm = n => sym + n.toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g,',');
  return { subtotal:sub, taxAmt, total, tax, sym, fm };
}

  const fN=V('fromName')||'Your Business', fT=V('fromTagline'), fE=V('fromEmail'), fA=V('fromAddress');

  const { subtotal, taxAmt, total, tax, fm } = calcTotals();
  const sL = { draft:'Draft', sent:'Sent', paid:'Paid ✓', overdue:'Overdue' };
  const fromD = [fE, fA].filter(Boolean).join('\n');

  }).join('');
  document.getElementById('invoicePreview').innerHTML = `

        ${fT?`<div class="inv-logo-tagline">${esc(fT)}</div>`:''}

      </div>
      <div class="inv-title-block">
        <div class="inv-title">INVOICE</div>
        <div class="inv-number">${esc(iN)}</div>
        <div style="margin-top:9px"><span class="status-badge status-${st}">${sL[st]}</span></div>
      </div>
    </div>
    <div class="inv-parties">
      <div>
        <div class="inv-party-label">Bill To</div>
        <div class="inv-party-name">${esc(tN)}</div>
        ${toD?`<div class="inv-party-detail">${esc(toD)}</div>`:''}
      </div>
      <div>
        <div class="inv-meta">
          <div class="inv-meta-item"><span class="inv-meta-key">Issued</span><span class="inv-meta-val">${fmtD(iD)}</span></div>
          <div class="inv-meta-item"><span class="inv-meta-key">Due</span><span class="inv-meta-val">${fmtD(dD)}</span></div>
